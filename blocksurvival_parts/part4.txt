function sanitizeLevel(o){
  if(!o||typeof o!=='object'||o.fmt!=='blocksurvival-level'||!Array.isArray(o.b))throw new Error("That doesn't look like a BlockSurvival level.");
  const num=(v,a,b,d)=>{v=Math.round(Number(v));return Number.isFinite(v)?clamp(v,a,b):d;};
  const isI=(v,a,b)=>Number.isInteger(v)&&v>=a&&v<=b;
  const s=o.set&&typeof o.set==='object'?o.set:{},iv=o.inv&&typeof o.inv==='object'?o.inv:{};
  const L={fmt:'blocksurvival-level',v:1,name:cleanName(o.name)||'Shared Level',size:SIZES[o.size]?o.size:'medium',
    set:{nights:num(s.nights,1,10,3),day:DAYS[s.day]?s.day:'normal',diff:DIFF[s.diff]?s.diff:'normal',zombies:num(s.zombies,0,15,3),more:num(s.more,0,4,1)},inv:{},b:[],s:[],vil:[],zsp:[]};
  for(const id of INV_IDS){const n=num(iv[id],0,99,0);if(n)L.inv[id]=n;}
  const seen=new Set();
  for(let i=0;i+4<o.b.length&&L.b.length<5*40000;i+=5){const [x,y,z,t,r]=o.b.slice(i,i+5);if(!isI(x,LX0,LX1)||!isI(z,LX0,LX1)||!isI(y,LY0,LY1)||!isI(t,0,BT.length-1))continue;
    const k=key(x,y,z);if(seen.has(k))continue;seen.add(k);L.b.push(x,y,z,t,r===1?1:0);}
  const pts=(a,max)=>{const out=[],sn=new Set();if(Array.isArray(a))for(const p of a){if(out.length>=max)break;if(!Array.isArray(p)||!isI(p[0],LX0,LX1)||!isI(p[1],LX0,LX1))continue;const k=p[0]+','+p[1];if(sn.has(k))continue;sn.add(k);out.push([p[0],p[1]]);}return out;};
  if(Array.isArray(o.s))for(const q of o.s){if(L.s.length>=400)break;if(Array.isArray(q)&&isI(q[0],0,SITEK.length-1)&&isI(q[1],LX0,LX1)&&isI(q[2],LY0,LY1)&&isI(q[3],LX0,LX1))L.s.push([q[0],q[1],q[2],q[3]]);}
  L.vil=pts(o.vil,12);L.zsp=pts(o.zsp,16);return L;}

function b64enc(u8){let s='';for(let i=0;i<u8.length;i+=0x8000)s+=String.fromCharCode.apply(null,u8.subarray(i,i+0x8000));return btoa(s).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');}
function b64dec(str){str=str.replace(/-/g,'+').replace(/_/g,'/');while(str.length%4)str+='=';const s=atob(str),u=new Uint8Array(s.length);for(let i=0;i<s.length;i++)u[i]=s.charCodeAt(i);return u;}
async function pipeBytes(u8,stream){return new Uint8Array(await new Response(new Blob([u8]).stream().pipeThrough(stream)).arrayBuffer());}
async function encodeLevel(L){const raw=new TextEncoder().encode(JSON.stringify(L));
  if(typeof CompressionStream!=='undefined'){try{return 'BSL1Z-'+b64enc(await pipeBytes(raw,new CompressionStream('deflate-raw')));}catch(e){}}
  return 'BSL1-'+b64enc(raw);}
async function decodeLevel(text){text=String(text||'').trim();if(!text)throw new Error('Paste a level code first (or load a file).');
  let json;
  if(text[0]==='{')json=text;
  else{const m=text.replace(/\s+/g,'').match(/^BSL1(Z?)-([A-Za-z0-9_-]+)$/);if(!m)throw new Error("That code doesn't look right. Level codes start with BSL1.");
    try{let u=b64dec(m[2]);if(m[1]){if(typeof DecompressionStream==='undefined')throw 0;u=await pipeBytes(u,new DecompressionStream('deflate-raw'));}json=new TextDecoder().decode(u);}
    catch(e){throw new Error('That code seems broken or cut off. Try copying all of it again.');}}
  let o;try{o=JSON.parse(json);}catch(e){throw new Error('That level seems broken or cut off.');}
  return sanitizeLevel(o);}
