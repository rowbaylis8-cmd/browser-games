  for(const [r,list] of byRegion){const R=regions[r];list.forEach((v,k)=>{const ci=R.cells[Math.floor((k+.5)*R.cells.length/list.length)];const j=list.length>R.cells.length?(hash(k,3)-.5)*.5:0;v.spot=[gxOf(ci)+.5+j,gzOf(ci)+.5+j];});}
  for(const v of alive){
    let dx=0,dz=0,sp=0,mode='walk';
    const nd=nearestZombieD(v.x,v.z);
    if(want){
      if(v.sheltered){mode='safe';if(v.spot){const ex=v.spot[0]-v.x,ez=v.spot[1]-v.z,d=Math.hypot(ex,ez);if(d>.2){dx=ex/d;dz=ez/d;sp=WALK*.8;}}}

      else{const f=fleeDir(v);if(f){mode='panic';dx=f[0];dz=f[1];sp=RUN;}else mode='worried';}

      if(v.idle>0)v.idle-=dt;

          else{const tg=flowTarget(v.field,passV,v.x,v.z);if(tg){const ex=tg[0]-v.x,ez=tg[1]-v.z,d=Math.hypot(ex,ez)||1;dx=ex/d;dz=ez/d;sp=WALK;}else{v.goal=-1;v.idle=.5;}}}}
      if(mode==='worried')sp*=.75;
    }
    let sx=0,sz=0;for(const o of alive){if(o===v)continue;const ex=v.x-o.x,ez=v.z-o.z,d=Math.hypot(ex,ez);if(d<.6&&d>1e-4){sx+=ex/d*(.6-d)*4;sz+=ez/d*(.6-d)*4;}}
    const ox=v.x,oz=v.z;moveEnt(v,dx*sp+sx,dz*sp+sz,dt,passV);
    v.spd=Math.hypot(v.x-ox,v.z-oz)/dt;v.anim+=dt*v.spd*3.3;v.mode=mode;
