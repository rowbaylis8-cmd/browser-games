// ------------------------------------------------------------------ autoplay demo bot: gathers, crafts and builds the hut, bonks zombies
function findSource(type){let best=null,bs=1e9;
  for(const [k,b] of world){if(b.type!==type||HUTSET.has(k))continue;if((type==='log'||type==='rock')&&!(b.nat>=0))continue;
    const [x,y,z]=k.split(',').map(Number);if(y<-1||(y===-1&&occupiedCell(x,z)))continue;
    const d=Math.hypot(x+.5-3.5,z+.5-2.5)-y*.05+(y<0?3:0);if(d<bs){bs=d;best={x,y,z,k};}}
  return best;}
function botNeed(item){const r=RECIPES.find(r=>r.out===item);
  if(r){for(const [k,n] of Object.entries(r.need))if(inv[k]<n){botNeed(k);return;}bot.act={kind:'craft',r,t:0,dur:.3};return;}
  const src={log:'log',stone:'rock',sand:'sand'}[item],s=src&&findSource(src);if(!s){bot.cool=1;return;}
  bot.act={kind:'gather',...s,t:0,dur:MINE_T[src]*.9};sel=0;}
function botStep(dt){
  if(bot.act){const a=bot.act;a.t+=dt;
    if(a.kind==='gather'){mining={x:a.x,y:a.y,z:a.z,p:clamp(a.t/a.dur,0,1)};if(a.t>=a.dur){mining=null;bot.act=null;if(world.has(a.k))gatherBlock(a.x,a.y,a.z,true);}}
    else if(a.t>=a.dur){bot.act=null;if(a.kind==='craft')craft(a.r,1);else if(a.kind==='place')placeBlock(a.x,a.y,a.z,a.item,0);}
    return;}
  if(bot.cool>0){bot.cool-=dt;return;}
  let tgt=null,bd=5;for(const zb of zombies){if(zb.dead||zb.rise<1||zb.burn>=0)continue;for(const v of villagers){if(!v.alive||v.sheltered)continue;const d=Math.hypot(v.x-zb.x,v.z-zb.z);if(d<bd){bd=d;tgt=zb;}}}
  if(tgt){bonk(tgt,true);bot.cool=.75;return;}
  while(bot.i<HUT.length){const e=HUT[bot.i],b=world.get(key(e.x,e.y,e.z));if(b&&b.type===e.type){bot.i++;continue;}break;}
  if(bot.i>=HUT.length){bot.cool=.5;return;}
  const e=HUT[bot.i],k=key(e.x,e.y,e.z);
  if(world.has(k)){const b=world.get(k);bot.act={kind:'gather',x:e.x,y:e.y,z:e.z,k,t:0,dur:MINE_T[b.type]||.4};return;}
  if(inv[e.type]>0){if(e.type!=='torch'&&e.y<=1&&occupiedCell(e.x,e.z)){bot.cool=.25;return;}bot.act={kind:'place',x:e.x,y:e.y,z:e.z,item:e.type,t:0,dur:.3};sel=IIDX[e.type];return;}
  botNeed(e.type);
}

// ------------------------------------------------------------------ simulation step
function step(dt){
  const pt=simT;simT+=dt;
  const d0=Math.floor(pt/CYCLE),d1=Math.floor(simT/CYCLE),c0=cyc(pt),c1=cyc(simT);
  if(d1>d0)onDawn(d0+1);
  else{if(c0<DUSKWARN&&c1>=DUSKWARN)onDuskWarn();if(c0<DAYLEN&&c1>=DAYLEN)onNight(d1+1);}
  if(navDirty)computeNav();
  if(isNight(simT)&&spawnLeft>0&&simT>=nextSpawn&&c1<CYCLE-8){spawnZombie();spawnLeft--;nextSpawn=simT+(spawnQuota-spawnLeft<2?1.3:4.5);}
  zTimer-=dt;if(zTimer<=0){zTimer=.3;computeZField();}
  updateVillagers(dt);updateZombies(dt);
  if(newVillAt!==null&&simT>=newVillAt){newVillAt=null;if(villagers.filter(v=>v.alive).length<VSP.length)addVillager(NEWV[0],NEWV[1],true);}
  if(AUTO)botStep(dt);
  fx=fx.filter(f=>simT-f.t<(f.life||.55)+.05);
  if(!gameOver&&villagers.every(v=>!v.alive)){gameOver=true;gameOverT=simT;onGameOver();}
  if(gameOver&&AUTO&&simT-gameOverT>5)resetGame();
}
function advance(sec){const n=Math.ceil(sec*30);for(let i=0;i<n;i++)step(sec/n);}
